#[cfg(windows)]
#[tokio::test]
async fn audit_direct_executable_can_receive_its_original_arguments() {
    let (program, args) = launch_command("node", &["--version".to_string()]);
    let output = Command::new(program).args(args).output().await.unwrap();
    assert!(output.status.success(), "{}", String::from_utf8_lossy(&output.stderr));
    assert!(String::from_utf8_lossy(&output.stdout).trim().starts_with('v'));
}
